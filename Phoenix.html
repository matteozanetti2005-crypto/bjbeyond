async function fetchAndAnalyze() { const url = document.getElementById('postUrl').value.trim(); if (!url) return alert("Please enter a post URL"); document.getElementById('resultsPanel').style.display = 'none'; document.querySelectorAll('.pipeline-step').forEach(el => el.classList.remove('active')); try { const proxyUrl = `https://api.allorigins.win/raw?url=${encodeURIComponent(url)}`; const response = await fetch(proxyUrl); const html = await response.text(); let postText = ''; // === TENTATIVO 1: Meta og:description === let match = html.match(/ 10) postText = match[1]; // === TENTATIVO 2: Title tag === if (!postText || postText.length < 15) { match = html.match(/([^<]+)<\/title>/); if (match && match[1]) postText = match[1].replace(' | X', '').replace(' on X', ''); } // === TENTATIVO 3: Pattern "text" (molto comune su X) === if (!postText || postText.length < 15) { match = html.match(/"text":"([^"]{15,})"/); if (match && match[1]) postText = match[1]; } // === TENTATIVO 4: data-testid="tweetText" === if (!postText || postText.length < 15) { match = html.match(/data-testid="tweetText"[^>]*>([^<]{15,})/); if (match && match[1]) postText = match[1]; } // === TENTATIVO 5: Qualsiasi span con tanto testo === if (!postText || postText.length < 15) { const spans = html.match(/<span[^>]*>([^<]{20,})<\/span>/g); if (spans) { for (let span of spans) { const text = span.replace(/<[^>]+>/g, ''); if (text.length > 20 && text.length < 500) { postText = text; break; } } } } // Pulisci il testo postText = postText .replace(/\\u002F/g, '/') .replace(/\\"/g, '"') .replace(/&/g, '&') .replace(/\\n/g, ' ') .trim(); if (!postText || postText.length < 10) { alert("Non sono riuscito a estrarre il testo. Incolla manualmente."); return; } // Esegui simulazione const results = generatePhoenixResults(postText, currentPersona); lastResults = results; updateResultsUI(results); const steps = ['hydration','sourcing','prefilter','transformer','scoring','ranking']; for (let i = 0; i < steps.length; i++) { await new Promise(r => setTimeout(r, 180)); highlightStep(steps[i], true); } document.getElementById('resultsPanel').style.display = 'block'; } catch (error) { alert("Errore durante il recupero. Incolla il testo manualmente."); } }
